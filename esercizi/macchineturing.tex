\begin{ex}
    Dimostrare:

\begin{enumerate}
    \item Se $X$ Turing decidibile allora $A^* \setminus X$ è Turing decidibile;
    \item Se $X \in \mathrm{DTIME}(T(n))$ allora $A^* \setminus X \in \mathrm{DTIME}(T(n))$.
\end{enumerate}
\end{ex}
\begin{sol}\
    \begin{enumerate}
        \item Se $\mu$ è una macchina di Turing
        che decide $X$, allora possiamo modificarla e creare
        una macchina di Turing $\mu_1$ che è uguale a quella di partenza
        invertendo solamente gli stati $q_F^0$ con $q_F^1$.
        Definiamo quindi
        \[
        \begin{matrix}
            \mu_1 : & Q \times \tilde A 
            & \to & Q' \times \tilde A \times \{-1,1 \}
             \\ & 
            (q,a)
            &
            \mapsto 
            &
            \begin{cases}
                (p,b,m), & \text{se } \mu(q,a)=(p,b,m) \text{ e } p\neq q_F^0,q_F^1
                \\
                (q^1_F,b,m), & \text{se } \mu(q,a)=(p,b,m) \text{ e } p= q_F^0
                \\
                (q^0_F,b,m), & \text{se } \mu(q,a)=(p,b,m) \text{ e } p= q_F^1
            \end{cases}
        \end{matrix}
        \]
        Questa chiaramente decide $A^*\setminus X$.
        \item La macchina di turing $\mu_1$ costruita
        nel punto precedente fa esattamente lo stesso numero di passi
        che compie una macchina $\mu$.
        Di conseguenza, se
        una macchina
        $\mu$ decide $X$ in tempo $T$, allora $\mu'$
        decide $A^*\setminus X$ in tempo $T$.
    \end{enumerate}
\end{sol}

    \item Se $X \in \mathrm{DTIME}_2 (T_1(n))$ e

         \[
        \begin{matrix}
            \overline{\mu} : & Q \times \tilde A 
            & \to & Q' \times \tilde A \times \{-1,1 \}
             \\ & 
            (q,a)
            &
            \mapsto 
            &
            \begin{cases}
                \alpha(q,a), & \text{se } q=q_0
                \\
                (p,b,m), & \text{se } \mu_X(q,a)=(p,b,m) \text{ e } p\neq q_F^1

                \\
                (q^0_{F_Y},b,m)

        \end{matrix}
        \]
    \end{enumerate}

\begin{ex}
Sia $2^{[d]}(n)$ la torre di esponenziali definita da
\[
2^{[d]}(n)=
\begin{cases}
n & \text{se } d=0,\\[2pt]

Mostrare che
\[

\begin{sol}
Usiamo la relazione
\[
\mathrm{DTIME}_1^{\{0,1\}}(T)\ \overset{(a)}{\subseteq}\ \mathrm{DSPACE}_1^{\{0,1\}}(T)\ \overset{(b)}{\subseteq}\ \bigcup_{\varepsilon>0}\mathrm{DTIME}_1^{\{0,1\}}\big(2^{\varepsilon T}\big)
\]
Mostriamo la doppia inclusione.
\begin{enumerate}
\item[``$\subseteq$''] Segue da $(a)$ con $T(n)=2^{[d]}(n)$: per ogni $d\ge1$ si ha
\[
\mathrm{DTIME}_1^{\{0,1\}}\big(2^{[d]}(n)\big)\subseteq\mathrm{DSPACE}_1^{\{0,1\}}\big(2^{[d]}(n)\big),
\]
e passando all'unione su $d\ge1$ si ottiene l'inclusione.
\item[``$\supseteq$''] Sia $X\in\mathrm{DSPACE}_1^{\{0,1\}}\big(2^{[d]}(n)\big)$ per qualche $d\ge1$. Per la relazione $(b)$ con $T(n)=2^{[d]}(n)$, esiste $\varepsilon>0$ tale che
\[
X\in\mathrm{DTIME}_1^{\{0,1\}}\big(2^{\varepsilon\cdot2^{[d]}(n)}\big).
\]
Posto $t=2^{[d]}(n)$, 
per $n$ sufficientemente grande si ha 
$\varepsilon t\le 2^t$, quindi definitivamente valgono
le seguenti disuguaglianze
\[
2^{\varepsilon\cdot2^{[d]}(n)}\le 
2^{2^{2^{[d]}(n)}} =
2^{2^{[d+1]}(n)}
=2^{[d+2]}(n).
\]
Dunque $X\in\mathrm{DTIME}_1^{\{0,1\}}
\big(2^{[d+2]}(n)\big)$, 
che compare nell'unione di sinistra perché 
$d+2\ge1$.
\end{enumerate}
\end{sol}

\begin{ex}
Sia $A=\{0,1\}$ e sia $B\supseteq A$ un alfabeto con $|B|=10$. Un insieme $X\subseteq A^*$ è deciso da una macchina di Turing a $k\geq 2$ nastri di alfabeto $B$ in tempo $T(n)\geq 3n$. Mostrare che $X$ è deciso anche da una macchina a $k$ nastri di alfabeto $A$, e determinare una costante $c$ tale che il tempo sia al più $c\,T(n)$.
\end{ex}

\begin{sol}
Codifichiamo i simboli di $B$ con blocchi di lunghezza $d$ su $A$, e il blank con $\square^d$. Serve che i $|B|+1$ simboli di $B\cup\{\square\}$ abbiano immagini distinte tra le $(|A|+1)^d$ parole di lunghezza $d$ su $A\cup\{\square\}$, cioè $11\leq 3^d$. Poiché $3^2=9<11\leq 27=3^3$, il minimo è $d=3$. Il tempo è quindi al più $4d\,T(n)=12\,T(n)$, e si può prendere $c=12$.
\end{sol}

\begin{ex}
Sia $A=\{0,1\}$ e sia $B\supseteq A$ un alfabeto di lavoro. Si vuole simulare una macchina di alfabeto $B$ con una di alfabeto $A$, codificando ogni simbolo di $B$ (blank incluso) con un blocco di lunghezza fissata $d$ su $A\cup\{\square\}$, in modo che ogni simbolo di $A$ e il blank siano codificati da blocchi costanti ($0\mapsto 0^d$, $1\mapsto 1^d$, $\square\mapsto\square^d$). Per quali valori di $|B|$ basta $d=2$? E per quali serve $d=3$?
\end{ex}

\begin{sol}
Le parole di lunghezza $d$ su $A\cup\{\square\}$ sono $3^d$, e i simboli da codificare sono $|B|+1$ (compreso il blank). Basta quindi $d=2$ se $|B|+1\leq 9$ e $d=1$ non basta, cioè $3\leq|B|\leq 8$. Serve $d=3$ se $|B|+1\leq 27$ e $d=2$ non basta, cioè $9\leq|B|\leq 26$. In generale $d$ è il minimo intero con $3^d\geq|B|+1$.
\end{sol}

\begin{ex}
Sia $B\supseteq\{0,1\}$ con $|B|=5$, e sia $X\subseteq\{0,1\}^*$ deciso in tempo $T(n)$ da una macchina a $k\geq 2$ nastri di alfabeto $B$.
\begin{enumerate}
  \item Qual è la lunghezza minima $d$ dei blocchi se si vuole una codifica iniettiva di $B$ in $\{0,1\}^d$?
  \item Qual è la lunghezza minima dei blocchi se si accetta di usare anche il blank nelle parole di codifica, richiedendo che $0,1,\square$ abbiano come immagine un blocco costante? Confrontare i due casi.
\end{enumerate}
\end{ex}

\begin{sol}\
\begin{enumerate}
  \item Servono almeno $5$ parole distinte di $\{0,1\}^d$, cioè $2^d\geq 5$, quindi $d=3$.
  \item I simboli da codificare sono $|B|+1=6$ e le parole di lunghezza $d$ su $\{0,1,\square\}$ sono $3^d$, quindi $d=2$ (perché $6\leq 9$).
\end{enumerate}
Nel secondo caso il blocco è più corto, quindi la simulazione è più veloce, perché il blank dà un simbolo in più per ogni cella del blocco.
\end{sol}

\begin{ex}
Sia $A=\{0,1\}$ e sia $X\subseteq A^*$ deciso in tempo $T(n)=n^2$ da una macchina a un nastro di alfabeto $B\supseteq A$ con $|B|=8$. Stimare il tempo di una macchina a un nastro di alfabeto $A$ che decide $X$.
\end{ex}

\begin{sol}
Serve $9\leq 3^d$, quindi $d=2$. Con un solo nastro, prima di simulare bisogna riscrivere l'input $w$ come $\omega(w)$, e questo costa $2n^2$ passi, perché va spostata ogni volta a destra la parte restante della parola. Il tempo totale è quindi al più
\[
3d\,T(n)+2n^2=6n^2+2n^2=8n^2 .
\]
\end{sol}



\begin{ex}
Si vuole simulare una macchina di alfabeto $B\supseteq A$ con una di alfabeto $A$ tramite blocchi di lunghezza $d$. Spiegare perché la condizione da imporre su $d$ è $|B|+1\leq(|A|+1)^d$ e non $|B|\leq|A|^d$.
\end{ex}

\begin{sol}
Durante la computazione le celle del nastro possono contenere anche il blank, quindi i simboli da codificare sono quelli di $B\cup\{\square\}$, in numero $|B|+1$. I blocchi vanno costruiti sull'alfabeto $A\cup\{\square\}$, perché una cella vuota di $\mu$ corrisponde a $d$ celle vuote della nuova macchina, e quindi sono $(|A|+1)^d$. La codifica deve essere iniettiva, e questo dà la condizione.
\end{sol}
